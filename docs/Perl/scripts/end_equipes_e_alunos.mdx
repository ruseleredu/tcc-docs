```perl
#!/usr/bin/perl
use strict;
use warnings;
use DBI;
use Getopt::Std;
use POSIX qw(strftime);

# -----------------------------------------------------------------------------
# Encerra equipes e alunos definitivamente.
# -----------------------------------------------------------------------------
$|++;
my $crlf = "\n";
my $aplicativo = $0;


# definição para acesso do banco (valores default)
my $host         = "localhost";
my $database     = "DAELT";
my $usuario      = "";
my $senha        = "";
my $final_commit = 0;

my $data_fechamento = strftime "%Y-%02m-%02d", localtime;

# Recuperação dos parâmetros de entrada (via shell)
my %opcoes=();
getopts("whu:p:fd:", \%opcoes);

$crlf         = "\r\n"     if defined $opcoes{w};
$usuario      = $opcoes{u} if defined $opcoes{u};
$senha        = $opcoes{p} if defined $opcoes{p};
$final_commit = 1          if defined $opcoes{f};

$data_fechamento = $opcoes{d} if defined $opcoes{d};

if (defined $opcoes{h})
{
	print STDERR "$aplicativo [-w] [-u <usuario>] [-p <senha>] [-f]" .
	             "[-d <YYYY-MM-DD>]$crlf";
	exit 0;
}


# apresentação do programa
print STDERR "$aplicativo\::Encerra equipes e alunos definitivamente.$crlf";


# conexão com banco de dados
my ($dbHandle);
my ($dbStatement1, $dbStatement2, $dbStatement3, $dbStatement4, $dbStatement5,
    $dbStatement6, $dbStatement7, $dbStatement8, $dbStatement9);
my (@dbLinha);

$dbHandle = DBI->connect("DBI:mysql:database=$database;host=$host",
                         $usuario, $senha,
                         {PrintError => 0, AutoCommit => 0})
	or die "$aplicativo\::Falha de conexão ao banco de dados: $DBI::errstr";

$dbStatement1 = $dbHandle->prepare(
	"SELECT A.ID, B.ID_TIPO " .
	"FROM TCC.EQUIPES A, TCC.BANCAS_EQUIPES B " .
	"WHERE A.ID = ? AND A.ID = B.ID_EQUIPE AND A.DATA_FIM IS NULL AND " .
	      "B.DATA_FIM IS NULL;");
$dbStatement2 = $dbHandle->prepare(
	"SELECT A.ID, B.ID_DOCUMENTO " .
	"FROM TCC.EQUIPES A, TCC.EQUIPES_DOCUMENTOS B " .
	"WHERE A.ID = ? AND A.ID = B.ID_EQUIPE AND A.DATA_FIM IS NULL AND " .
	      "B.DATA IS NULL;");
$dbStatement3 = $dbHandle->prepare(
	"UPDATE TCC.EQUIPES_DISCIPLINAS " .
	"SET DATA_FIM = ? " .
	"WHERE ID_EQUIPE = ? AND DATA_FIM IS NULL;");
$dbStatement4 = $dbHandle->prepare(
	"UPDATE TCC.EQUIPES_ORIENTADORES " .
	"SET DATA_FIM = ? " .
	"WHERE ID_EQUIPE = ? AND DATA_FIM IS NULL;");
$dbStatement5 = $dbHandle->prepare(
	"UPDATE TCC.EQUIPES_ALUNOS " .
	"SET DATA_FIM = ? " .
	"WHERE ID_EQUIPE = ? AND DATA_FIM IS NULL;");
$dbStatement6 = $dbHandle->prepare(
	"UPDATE TCC.EQUIPES_ALUNOS A, DAELT.ALUNOS_DISCIPLINAS B " .
	"SET B.DATA_FIM = ? " .
	"WHERE A.ID_EQUIPE = ? AND A.ID_ALUNO = B.ID_ALUNO AND " .
	      "B.DATA_FIM IS NULL;");
$dbStatement7 = $dbHandle->prepare(
	"UPDATE TCC.EQUIPES_ALUNOS A, DAELT.ALUNOS B " .
	"SET B.DATA_FIM = ? " .
	"WHERE A.ID_EQUIPE = ? AND A.ID_ALUNO = B.ID AND B.DATA_FIM IS NULL;");
$dbStatement8 = $dbHandle->prepare(
	"UPDATE TCC.EQUIPES " .
	"SET DATA_FIM = ? " .
	"WHERE ID = ? AND DATA_FIM IS NULL;");
$dbStatement9 = $dbHandle->prepare(
	"UPDATE TCC.EQUIPES SET ATIVA = FALSE WHERE ID = ?;");


# definição de data de início para TODAS as equipes
my $equipe_datainicio = strftime "%Y-%02m-%02d", localtime;


# obtém nome da equipe
print STDERR "Equipe (fecha em: $data_fechamento) (CTRL+D aborta):";
while (<STDIN>)
{
	my @informacao = split(' ');

	my $informacaoSize = scalar(@informacao);
	(print STDERR "$aplicativo\::Número de parâmetros incorretos$crlf" and
	 $dbHandle->disconnect() and die)
		if ($informacaoSize != 1);

	my $equipe_id  = $informacao[0];

	# checagem de bancas abertas
	$dbStatement1->bind_param(1, $equipe_id);
	$dbStatement1->execute()
		or (print "$aplicativo\::Falha de checagem: " .
		          "$DBI::errstr$crlf" and
		    $dbHandle->disconnect() and die);

	my $checa_existencia = 0;
	while ( @dbLinha = $dbStatement1->fetchrow_array )
	{
		print "Existem bancas ainda abertas:$crlf"
			if $checa_existencia == 0;
		$checa_existencia = 1;

		print "$dbLinha[0], $dbLinha[1]$crlf";
	}

	if ($checa_existencia == 1)
	{
		$dbHandle->disconnect();
		die "Feche bancas abertas.";
	}

	# checagem de documentos abertas
	$dbStatement2->bind_param(1, $equipe_id);
	$dbStatement2->execute()
		or (print "$aplicativo\::Falha de checagem: " .
		          "$DBI::errstr$crlf" and
		    $dbHandle->disconnect() and die);

	while ( @dbLinha = $dbStatement2->fetchrow_array )
	{
		print "Existem documentos ainda abertos:$crlf"
			if $checa_existencia == 0;
		$checa_existencia = 1;

		print "$dbLinha[0], $dbLinha[1]$crlf";
	}

	if ($checa_existencia == 1)
	{
		$dbHandle->disconnect();
		die "Feche documentos abertos.";
	}

	# efetua fechamento equipes (disciplinas, orientadores, alunos)
	$dbStatement3->bind_param(1, $data_fechamento);
	$dbStatement3->bind_param(2, $equipe_id);
	$dbStatement3->execute()
		or (print "$aplicativo\::Falha de fechamento: " .
		          "$DBI::errstr$crlf" and
		    $dbHandle->disconnect() and die);

	$dbStatement4->bind_param(1, $data_fechamento);
	$dbStatement4->bind_param(2, $equipe_id);
	$dbStatement4->execute()
		or (print "$aplicativo\::Falha de fechamento: " .
		          "$DBI::errstr$crlf" and
		    $dbHandle->disconnect() and die);

	$dbStatement5->bind_param(1, $data_fechamento);
	$dbStatement5->bind_param(2, $equipe_id);
	$dbStatement5->execute()
		or (print "$aplicativo\::Falha de fechamento: " .
		          "$DBI::errstr$crlf" and
		    $dbHandle->disconnect() and die);

	# efetua fechamento dos alunos
	$dbStatement6->bind_param(1, $data_fechamento);
	$dbStatement6->bind_param(2, $equipe_id);
	$dbStatement6->execute()
		or (print "$aplicativo\::Falha de fechamento: " .
		          "$DBI::errstr$crlf" and
		    $dbHandle->disconnect() and die);

	$dbStatement7->bind_param(1, $data_fechamento);
	$dbStatement7->bind_param(2, $equipe_id);
	$dbStatement7->execute()
		or (print "$aplicativo\::Falha de fechamento: " .
		          "$DBI::errstr$crlf" and
		    $dbHandle->disconnect() and die);

	# encerra equipe
	$dbStatement8->bind_param(1, $data_fechamento);
	$dbStatement8->bind_param(2, $equipe_id);
	$dbStatement8->execute()
		or (print "$aplicativo\::Falha de fechamento: " .
		          "$DBI::errstr$crlf" and
		    $dbHandle->disconnect() and die);

	$dbStatement9->bind_param(1, $equipe_id);
	$dbStatement9->execute()
		or (print "$aplicativo\::Falha de fechamento: " .
		          "$DBI::errstr$crlf" and
		    $dbHandle->disconnect() and die);

	$dbHandle->commit() if $final_commit==0;

	print STDERR "Equipe (fecha em: $data_fechamento) (CTRL+D aborta):";
}


# desconexão com banco de dados
$dbHandle->commit() if $final_commit==1;
$dbHandle->disconnect();
exit 0;
```
