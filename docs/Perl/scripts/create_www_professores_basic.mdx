# -----------------------------------------------------------------------------
# Embaralha email para evitar spam!
# -----------------------------------------------------------------------------
sub embaralha_email
{
	my $email = $_[0];

	if ($email =~ /^([\w\.\-\_]+)\@([\w\.\-\_]+[a-zA-Z]{2,})$/)
	{
		my $nome = $1;
		my $host = $2;
		$host =~ s/\./ [dot] /g;
		$email = $nome . " [at] " . $host;
	}

	return $email;
}
