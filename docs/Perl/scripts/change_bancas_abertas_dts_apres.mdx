```perl
#!/usr/bin/perl
use strict;
use warnings;
use DBI;
use Getopt::Std;
use POSIX qw(strftime);

# -----------------------------------------------------------------------------
# Altera a nota de uma banca que esteja aberta.
# -----------------------------------------------------------------------------
$|++;
my $crlf = "\n";
my $aplicativo = $0;


# definição para acesso do banco (valores default)
my $host         = "localhost";
my $database     = "DAELT";
my $usuario      = "";
my $senha        = "";
my $final_commit = 0;


# Recuperação dos parâmetros de entrada (via shell)
my %opcoes=();
getopts("whu:p:f", \%opcoes);

$crlf         = "\r\n"     if defined $opcoes{w};
$usuario      = $opcoes{u} if defined $opcoes{u};
$senha        = $opcoes{p} if defined $opcoes{p};
$final_commit = 1          if defined $opcoes{f};

if (defined $opcoes{h})
{
	print STDERR "$aplicativo [-w] [-u <usuario>] [-p <senha>] [-f]$crlf";
	exit 0;
}


# apresentação do programa
print STDERR "$aplicativo\::Altera a nota de uma banca que esteja aberta.$crlf";


# conexão com banco de dados
my ($dbHandle, $dbStatement);

$dbHandle = DBI->connect("DBI:mysql:database=$database;host=$host",
                         $usuario, $senha,
                         {PrintError => 0, AutoCommit => 0})
	or die "$aplicativo\::Falha de conexão ao banco de dados: $DBI::errstr";

$dbStatement = $dbHandle->prepare(
	"UPDATE TCC.BANCAS_EQUIPES " .
	"SET DATA_APRESENTACAO = ?, HORA_APRESENTACAO = ? " .
	"WHERE ID_EQUIPE = ? AND ID_TIPO = ? AND DATA_FIM IS NULL;");


# definição de data de início para TODAS as bancas
my $banca_datainicio = strftime "%Y-%02m-%02d", localtime;


# obtém nome da equipe
print STDERR "Cód equipe <spc> tipo banca " .
             "(1-prop/2-defesa prop/3-tcc1/4-tcc2 <spc> " .
             "dt apres (YYYY-MM-DD) <spc> hr apres (HH-MM-SS ou NULL) " .
             "(CTRL+D aborta):";
while (<STDIN>)
{
	my @informacao = split(' ');

	my $informacaoSize = scalar(@informacao);
	(print STDERR "$aplicativo\::Número de parâmetros incorretos$crlf" and
	 $dbHandle->disconnect() and die)
		if ($informacaoSize != 4);

	(print STDERR "$aplicativo\::Data inválida$crlf" and
	 $dbHandle->disconnect() and die)
		if ($informacao[2] !~ /\d\d\d\d-\d\d-\d\d/);

	(print STDERR "$aplicativo\::Hora inválida$crlf" and
	 $dbHandle->disconnect() and die)
		if ($informacao[3] !~ /(\d\d-\d\d-\d\d)|NULL/);

	$informacao[3] =~ s/-/:/g if $informacao[3] ne 'NULL';
	$informacao[3] =  undef   if $informacao[3] eq 'NULL';

	$dbStatement->bind_param(1, $informacao[2]);
	$dbStatement->bind_param(2, $informacao[3]);
	$dbStatement->bind_param(3, $informacao[0]);
	$dbStatement->bind_param(4, $informacao[1]);

	$dbStatement->execute()
		or (print STDERR "$aplicativo\::Falha de alteracao: " .
		                 "$DBI::errstr$crlf" and
		    $dbHandle->disconnect() and die);

	$dbHandle->commit() if $final_commit==0;

	print STDERR "Cód equipe <spc> tipo banca " .
	             "(1-prop/2-defesa prop/3-tcc1/4-tcc2 <spc> " .
	             "dt apres (YYYY-MM-DD) <spc> hr apres (HH-MM-SS ou NULL) " .
	             "(CTRL+D aborta):";
}


# desconexão com banco de dados
$dbHandle->commit() if $final_commit==1;
$dbHandle->disconnect();
exit 0;
```
