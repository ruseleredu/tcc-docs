#!/usr/bin/perl
use strict;
use warnings;
use DBI;
use Getopt::Std;

# -----------------------------------------------------------------------------
# Apresenta listagem de professores.
# -----------------------------------------------------------------------------
$|++;
my $crlf = "\n";
my $aplicativo = $0;


# definição para acesso do banco (valores default)
my $host     = "localhost";
my $database = "DAELT";
my $usuario  = "";
my $senha    = "";
my $docente  = "";


# Recuperação dos parâmetros de entrada (via shell)
my %opcoes=();
getopts("whu:p:d:", \%opcoes);

$crlf    = "\r\n"     if defined $opcoes{w};
$usuario = $opcoes{u} if defined $opcoes{u};
$senha   = $opcoes{p} if defined $opcoes{p};
$docente = $opcoes{d} if defined $opcoes{d};

if (defined $opcoes{h})
{
	print STDERR "$aplicativo [-w] [-u <usuario>] [-p <senha>] " .
	             "[-d <codigo docente>]$crlf";
	exit 0;
}


# apresentação do programa
print STDERR "$aplicativo\::Gera arquivo PDF contendo certificados " .
             "de participacao de docentes (todas as épocas).$crlf";


# conexão com banco de dados
my ($dbHandle,
    $dbStatement1, $dbStatement2, $dbStatement3, $dbStatement4);
my (@dbLinha1, @dbLinha2);

$dbHandle = DBI->connect("DBI:mysql:database=$database;host=$host",
                         $usuario, $senha,
                         {PrintError => 0, AutoCommit => 0})
	or die "$aplicativo\::Falha de conexão ao banco de dados: $DBI::errstr";

$dbStatement1 = $dbHandle->prepare(
	"SELECT B1.ID, B1.TITULO, A1.ID, A1.NOME, B.ID, B.DATA_APRESENTACAO, " .
	       "B2.NOME " .
	"FROM TCC.BANCAS_PROFESSORES A, DAELT.PROFESSORES A1, " .
	     "TCC.BANCAS_EQUIPES B, TCC.EQUIPES B1, TCC.DOMINIO_BANCAS B2 " .
	"WHERE A.ID_PROFESSOR = ? AND A.ID_PROFESSOR = A1.ID AND " .
	      "A.ID_BANCA = B.ID AND B.ID_EQUIPE = B1.ID AND B.ID_TIPO = B2.ID " .
	"ORDER BY B.ID_TIPO, B.DATA_APRESENTACAO;");
$dbStatement2 = $dbHandle->prepare(
	"SELECT B.NOME " .
	"FROM TCC.EQUIPES_ALUNOS A, DAELT.ALUNOS B " .
	"WHERE A.ID_EQUIPE = ? AND A.ID_ALUNO = B.ID AND " .
	      "A.DATA_INICIO <= ? AND (A.DATA_FIM IS NULL OR A.DATA_FIM >= ?) " .
	"ORDER BY B.NOME;");
$dbStatement3 = $dbHandle->prepare(
	"SELECT B.CURSO, C1.NOME, C2.NOME " .
	"FROM TCC.EQUIPES_DISCIPLINAS A, DAELT.DOMINIO_DISCIPLINAS B, " .
	     "DAELT.PROFESSORES C1, DAELT.PROFESSORES C2 " .
	"WHERE A.ID_EQUIPE = ? AND A.ID_DISCIPLINA = B.ID AND " .
	      "B.ID_RESPONSAVEL = C1.ID AND B.ID_COORDENADOR = C2.ID AND " .
	      "A.DATA_INICIO <= ? AND (A.DATA_FIM IS NULL OR A.DATA_FIM >= ?);");
$dbStatement4 = $dbHandle->prepare(
	"SELECT B.NOME " .
	"FROM TCC.BANCAS_PROFESSORES A, DAELT.PROFESSORES B " .
	"WHERE A.ID_BANCA = ? AND A.ID_PROFESSOR != ? AND A.ID_PROFESSOR = B.ID " .
	"ORDER BY B.NOME;");


# declaracao de algumas funcoes
sub get_data_extenso;
sub get_data_normal;


# impressão do preâmbulo (conforme página)
# top=45mm para alguns formulários e 53mm para outros
print <<PREAMBULO_DOCUMENTO;
\\documentclass[11pt,a4paper,notitlepage]{article}

\\usepackage[brazil]{babel}          % pacote de hifenação para Brasil
\\usepackage[latin1,utf8]{inputenc}  % tipo de caracter (acentuação)
\\usepackage[T1]{fontenc}            % codificação de fontes

\\usepackage{fullpage}
\\usepackage{amsmath}
\\usepackage{amsfonts}
\\usepackage{amssymb}
\\usepackage{graphicx}
\\usepackage{icomma}
\\usepackage{float}    % permite [H] de figure funcionar como deve!
\\usepackage[top=53mm, bottom=20mm, left=55mm, right=15mm]{geometry}
\\usepackage{setspace}

\\pagestyle{empty}

\\begin{document}

PREAMBULO_DOCUMENTO

# imprime 1 página por orientador/co-orientador
$dbStatement1->bind_param(1, $docente);
$dbStatement1->execute();

my $checa_existencia = 0;
while ( @dbLinha1 = $dbStatement1->fetchrow_array )
{
	print "$crlf\\newpage$crlf$crlf" if $checa_existencia == 1;
	$checa_existencia = 1;

	# ajuste de datas
	my $data = $dbLinha1[5];

	my $data_extenso = get_data_extenso($data);
	my $data_normal  = get_data_normal($data);

print <<PREAMBULO_PAGINA;
% ------------------------------- nova seção
\\begin{center}
\\bfseries
Departamento Acadêmico de Eletrotécnica \\\\
Trabalho de Conclusão de Curso
\\end{center}

\\vspace{10mm}

\\begin{center}
\\bfseries
\\scshape
\\Large
Declaração
\\end{center}

\\vspace{10mm}

\\begin{doublespace}
Declaramos que

\\begin{center}
\\Large
\\itshape
$dbLinha1[3]
\\end{center}

PREAMBULO_PAGINA

	print "participou da banca de avaliação de $dbLinha1[6] de Trabalho de " .
	      "Conclusão de Curso intitulado ``$dbLinha1[1]''";

	$dbStatement2->bind_param(1, $dbLinha1[0]);
	$dbStatement2->bind_param(2, $dbLinha1[5]);
	$dbStatement2->bind_param(3, $dbLinha1[5]);
	$dbStatement2->execute();

	my @lista_alunos;
	while ( @dbLinha2 = $dbStatement2->fetchrow_array )
	{
		push @lista_alunos, $dbLinha2[0];
	}

	my $lista_alunosSize = scalar(@lista_alunos);
	if ($lista_alunosSize==1 )
	{
		print " realizado pelo aluno $lista_alunos[0]";
	}
	elsif($lista_alunosSize!=0)
	{
		print " realizado pelos alunos $lista_alunos[0]";
		for (my $i=1; $i<$lista_alunosSize - 1; $i++)
		{
			print ", $lista_alunos[$i]";
		}
		print " e $lista_alunos[$lista_alunosSize - 1]";
	}

	print ", apresentado em $data_normal.$crlf";

	$dbStatement4->bind_param(1, $dbLinha1[4]);
	$dbStatement4->bind_param(2, $dbLinha1[2]);
	$dbStatement4->execute();

	my @lista_banca;
	while ( @dbLinha2 = $dbStatement4->fetchrow_array )
	{
		push @lista_banca, $dbLinha2[0];
	}

	my $lista_bancaSize = scalar(@lista_banca);
	if ($lista_bancaSize==1 )
	{
		print "$crlf";
		print "Fez parte da banca: $lista_banca[0].$crlf";
	}
	elsif ($lista_bancaSize!=0)
	{
		print "$crlf";
		print "Fizeram parte da banca: $lista_banca[0]";
		for (my $i=1; $i<$lista_bancaSize - 1; $i++)
		{
			print ", $lista_banca[$i]";
		}
		print " e $lista_banca[$lista_bancaSize - 1].$crlf";
	}

	$dbStatement3->bind_param(1, $dbLinha1[0]);
	$dbStatement3->bind_param(2, $dbLinha1[5]);
	$dbStatement3->bind_param(3, $dbLinha1[5]);
	$dbStatement3->execute();

	@dbLinha2 = $dbStatement3->fetchrow_array;
	while ( $dbStatement3->fetchrow_array )
	{
	}

print <<PERORACAO_PAGINA;

\\begin{flushright}
Curitiba, $data_extenso.
\\end{flushright}
\\end{doublespace}

\\vspace{10mm}

\\begin{center}
\\rule{9cm}{0.5pt}\\\\
$dbLinha2[2]\\\\
\\footnotesize
Coordenador do Curso de $dbLinha2[0]
\\end{center}

\\vspace{10mm}

\\begin{center}
\\rule{9cm}{0.5pt}\\\\
$dbLinha2[1]\\\\
\\footnotesize
Responsável pelos Trabalhos de Conclusão de Curso\\\\
$dbLinha2[0]
\\end{center}
PERORACAO_PAGINA
}


# impressão da peroracao (conforme página)
print <<PERORACAO_DOCUMENTO;

\\end{document}
PERORACAO_DOCUMENTO

# desconexão com banco de dados
$dbHandle->disconnect();
exit 0;


# -----------------------------------------------------------------------------
# funções de manipulação de data
# -----------------------------------------------------------------------------
sub get_data_extenso()
{
	my $data = $_[0];
	my @informacao = split('-', $data);
	my $mes;

	$mes = "janeiro"   if $informacao[1] == 1;
	$mes = "fevereiro" if $informacao[1] == 2;
	$mes = "março"     if $informacao[1] == 3;
	$mes = "abril"     if $informacao[1] == 4;
	$mes = "maio"      if $informacao[1] == 5;
	$mes = "junho"     if $informacao[1] == 6;
	$mes = "julho"     if $informacao[1] == 7;
	$mes = "agosto"    if $informacao[1] == 8;
	$mes = "setembro"  if $informacao[1] == 9;
	$mes = "outubro"   if $informacao[1] == 10;
	$mes = "novembro"  if $informacao[1] == 11;
	$mes = "dezembro"  if $informacao[1] == 12;

	return "$informacao[2] de $mes de $informacao[0]";
}


sub get_data_normal()
{
	my $data = $_[0];
	my @informacao = split('-', $data);

	return "$informacao[2]/$informacao[1]/$informacao[0]";
}
