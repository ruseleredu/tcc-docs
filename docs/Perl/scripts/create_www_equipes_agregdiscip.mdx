# Recuperação dos parâmetros de entrada (via shell)

$equipe_agregdiscip = $opcoes{a} if defined $opcoes{a};

if (defined $opcoes{h})

	print STDERR "$aplicativo [-w] [-u <usuario>] [-p <senha>] ".
	             "[-a <agregador de disciplina>]$crlf";
	exit 0;
}

                         $usuario, $senha,
                         {PrintError => 0, AutoCommit => 0})
	or die "$aplicativo\::Falha de conexão ao banco de dados: $DBI::errstr";

$dbStatement1 = $dbHandle->prepare(
	"SELECT A.ID, A.TITULO, B.NOME " .
	"FROM TCC.EQUIPES A, DAELT.PROFESSORES B, TCC.EQUIPES_DISCIPLINAS C, " .
	     "DAELT.DOMINIO_DISCIPLINAS D " .
	"WHERE A.ID !='0' AND A.ID_LIDER = B.ID AND A.ID = C.ID_EQUIPE AND " .
	      "A.DATA_FIM IS NULL AND C.DATA_FIM IS NULL AND " .
	      "C.ID_DISCIPLINA = D.ID AND D.ID_AGREGADOR = ? " .
	"ORDER BY A.ID;");
$dbStatement2 = $dbHandle->prepare(
	"SELECT ID_DISCIPLINA, ID_TURMA, B.DATA_FIM " .
	"FROM TCC.EQUIPES A, TCC.EQUIPES_DISCIPLINAS B " . 
	"WHERE A.ID = ? AND A.ID = B.ID_EQUIPE " .
	"ORDER BY B.DATA_INICIO;");
$dbStatement3 = $dbHandle->prepare(
	"SELECT C.NOME, B.DATA_FIM " .
	"FROM TCC.EQUIPES A, TCC.EQUIPES_ALUNOS B, DAELT.ALUNOS C " .
	"WHERE A.ID = ? AND A.ID = B.ID_EQUIPE AND B.ID_ALUNO = C.ID " .
	"ORDER BY C.NOME;");
$dbStatement4 = $dbHandle->prepare(
	"SELECT B.ORIENTACAO, C.NOME, B.DATA_FIM " .
	"FROM TCC.EQUIPES A, TCC.EQUIPES_ORIENTADORES B, DAELT.PROFESSORES C " . 
	"WHERE A.ID = ? AND A.ID = B.ID_EQUIPE " .
	               "AND B.ID_ORIENTADOR = C.ID;");
$dbStatement5 = $dbHandle->prepare(
	"SELECT A.ID, A.ID_DOCUMENTO, B.NOME, " .
	       "DATE_FORMAT(A.DATA_LIMITE, $data_formato_sql), A.DATA_LIMITE, " .
	       "DATE_FORMAT(A.DATA, $data_formato_sql), " .
	       "FORCADO " .
	"FROM TCC.EQUIPES_DOCUMENTOS A, TCC.DOCUMENTOS B " .
	"WHERE A.ID_DOCUMENTO = B.ID AND A.ID_EQUIPE = ? " .
	"ORDER BY A.DATA, A.DATA_LIMITE, A.ID_DOCUMENTO;");
$dbStatement6 = $dbHandle->prepare(
	"SELECT A.ID, B.NOME, FORMAT(A.NOTA, 1) " .
	"FROM TCC.BANCAS_EQUIPES A, TCC.DOMINIO_BANCAS B " .
	"WHERE A.ID_TIPO = B.ID AND A.ID_EQUIPE = ? " .
	"ORDER BY ID_TIPO, DATA_APRESENTACAO, DATA_INICIO;");
$dbStatement7 = $dbHandle->prepare(
	"SELECT B.NOME " .
	"FROM TCC.BANCAS_PROFESSORES A, DAELT.PROFESSORES B " .
	"WHERE A.ID_BANCA = ? AND A.ID_PROFESSOR = B.ID ".
	"ORDER BY B.NOME;");

while ( @dbLinha1 = $dbStatement1->fetchrow_array )

			if $dbLinha2[1] == 104 and

		# lógica conforme ordem de apresentação na lista
		$checa_necessidade_monografia = 1
			if $dbLinha2[1] == 313 and
			   !defined $dbLinha2[5] and $dbLinha2[4] ge $dataatual;
		$checa_necessidade_monografia = 2
			if $checa_necessidade_monografia == 1 and
			   $dbLinha2[1] == 303 and

		$checa_monografia_final = 1
			if $dbLinha2[1] == 313 and defined $dbLinha2[5];

			if !defined $dbLinha2[5] and $dbLinha2[4] lt $dataatual;
		print "\t\t<li class=\"tcc-doc-pending\"><b>$dbLinha2[3]</b>: " .

			if !defined $dbLinha2[2];
