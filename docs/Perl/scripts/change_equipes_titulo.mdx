#!/usr/bin/perl
use strict;
use warnings;
use DBI;
use Getopt::Std;
use POSIX qw(strftime);

# -----------------------------------------------------------------------------
# Altera dados da equipe de tcc em tabela.
# NÃO associa alunos nem orientadores.
# -----------------------------------------------------------------------------
$|++;
my $crlf = "\n";
my $aplicativo = $0;


# definição para acesso do banco (valores default)
my $host         = "localhost";
my $database     = "DAELT";
my $usuario      = "";
my $senha        = "";
my $final_commit = 0;


# Recuperação dos parâmetros de entrada (via shell)
my %opcoes=();
getopts("whu:p:f", \%opcoes);

$crlf         = "\r\n"     if defined $opcoes{w};
$usuario      = $opcoes{u} if defined $opcoes{u};
$senha        = $opcoes{p} if defined $opcoes{p};
$final_commit = 1          if defined $opcoes{f};

if (defined $opcoes{h})
{
	print STDERR "$aplicativo [-w] [-u <usuario>] [-p <senha>] [-f]$crlf";
	exit 0;
}


# apresentação do programa
print STDERR "$aplicativo\::Altera dados da equipe de tcc em tabela.$crlf";
print STDERR "$aplicativo\::NÃO associa alunos nem orientadores.$crlf";


# conexão com banco de dados
my ($dbHandle, $dbStatement);

$dbHandle = DBI->connect("DBI:mysql:database=$database;host=$host",
                         $usuario, $senha,
                         {PrintError => 0, AutoCommit => 0})
	or die "$aplicativo\::Falha de conexão ao banco de dados: $DBI::errstr";

$dbStatement = $dbHandle->prepare(
	"UPDATE TCC.EQUIPES SET TITULO = ? WHERE ID = ?;");


# definição de data de início para TODAS as equipes
my $equipe_datainicio = strftime "%Y-%02m-%02d", localtime;


# obtém nome da equipe
print STDERR "Cód equipe <spc> 'tít trabalho' (CTRL+D aborta):";
while (<STDIN>)
{
	(print STDERR "$aplicativo\::Número de parâmetros incorretos$crlf" and
	 $dbHandle->disconnect() and die)
		if (!/\s*(\S+)\s*'(.*)'\s*/);

	my $equipe_id     = $1;
	my $equipe_titulo = $2;

	$equipe_titulo =~ s/^\s*(.*?)\s*$/$1/;
	(print STDERR "$aplicativo\::Título inválido$crlf" and
	 $dbHandle->disconnect() and die)
		if length($equipe_titulo)==0;

	$dbStatement->bind_param(1, $equipe_titulo);
	$dbStatement->bind_param(2, $equipe_id);

	$dbStatement->execute()
		or (print STDERR "$aplicativo\::Falha de alteracao: " .
		                 "$DBI::errstr$crlf" and
		    $dbHandle->disconnect() and die);

	$dbHandle->commit() if $final_commit==0;

	print STDERR "Cód equipe <spc> 'tít trabalho' (CTRL+D aborta):";
}


# desconexão com banco de dados
$dbHandle->commit() if $final_commit==1;
$dbHandle->disconnect();
exit 0;
