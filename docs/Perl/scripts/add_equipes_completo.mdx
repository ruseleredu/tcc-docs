```perl
#!/usr/bin/perl
use strict;
use warnings;
use DBI;
use Getopt::Std;
use POSIX qw(strftime);

# -----------------------------------------------------------------------------
# Insere novo grupo de tcc em tabela.
# Associa também os alunos e orientadores.
# -----------------------------------------------------------------------------
$|++;
my $crlf = "\n";
my $aplicativo = $0;


# definição para acesso do banco (valores default)
my $host         = "localhost";
my $database     = "DAELT";
my $usuario      = "";
my $senha        = "";
my $final_commit = 0;

my $equipe_disciplina = "";
my $equipe_turma      = "";


# Recuperação dos parâmetros de entrada (via shell)
my %opcoes=();
getopts("whu:p:f1:2:", \%opcoes);

$crlf         = "\r\n"     if defined $opcoes{w};
$usuario      = $opcoes{u} if defined $opcoes{u};
$senha        = $opcoes{p} if defined $opcoes{p};
$final_commit = 1          if defined $opcoes{f};

$equipe_disciplina = $opcoes{1} if defined $opcoes{1};
$equipe_turma      = $opcoes{2} if defined $opcoes{2};

if (defined $opcoes{h})
{
	print STDERR "$aplicativo [-w] [-u <usuario>] [-p <senha>] [-f] " .
	             "[-1 <disciplina>] [-2 <turma>]$crlf";
	exit 0;
}


# apresentação do programa
print STDERR "$aplicativo\::Insere novo grupo de tcc em tabela.$crlf";
print STDERR "$aplicativo\::Associa também os alunos e orientadores.$crlf";


# conexão com banco de dados
my ($dbHandle);
my ($dbStatement1, $dbStatement2, $dbStatement3, $dbStatement4, $dbStatement5);

$dbHandle = DBI->connect("DBI:mysql:database=$database;host=$host",
                         $usuario, $senha,
                         {PrintError => 0, AutoCommit => 0})
	or die "$aplicativo\::Falha de conexão ao banco de dados: $DBI::errstr";

$dbStatement1 = $dbHandle->prepare(
	"INSERT INTO TCC.EQUIPES " .
	"(ID, TITULO, ATIVA, ID_LIDER, DATA_INICIO) VALUES (?, ?, TRUE, ?, ?);");
$dbStatement2 = $dbHandle->prepare(
	"INSERT INTO TCC.EQUIPES_DISCIPLINAS " .
	"(ID_EQUIPE, ID_DISCIPLINA, ID_TURMA, DATA_INICIO) VALUES (?, ?, ?, ?);");
$dbStatement3 = $dbHandle->prepare(
	"INSERT INTO TCC.EQUIPES_ALUNOS (ID_EQUIPE, ID_ALUNO, DATA_INICIO) " .
	"VALUES (?, ?, ?);");
$dbStatement4 = $dbHandle->prepare(
	"INSERT INTO TCC.EQUIPES_ORIENTADORES " .
	"(ID_EQUIPE, ID_ORIENTADOR, ORIENTACAO, DATA_INICIO) " .
	"VALUES (?, ?, ?, ?);");
$dbStatement5 = $dbHandle->prepare(
	"INSERT INTO TCC.EQUIPES_DOCUMENTOS " .
	"(ID_EQUIPE, ID_DOCUMENTO, DATA_LIMITE, FORCADO) " .
	"SELECT ?, ID_DOCUMENTO, DATA_LIMITE_ATUAL, FALSE " .
	"FROM TCC.AGREGADOR_DOCUMENTOS WHERE ID_AGREGADOR = ?;");


# definição de data de início para TODAS as equipes
my $equipe_datainicio = strftime "%Y-%02m-%02d", localtime;


# obtém nome da equipe
print STDERR "Cód ($equipe_disciplina/$equipe_turma) <spc> 'tít' <spc> " . 
             "id lider <spc> cód agregador <spc> núm alunos <spc> " .
             "aluno ... <spc> núm orientadores <spc> orientador <spc> " .
             "(0-Co/1-Ori) ... (CTRL+D aborta):";
while (<STDIN>)
{
	(print STDERR "$aplicativo\::Número de parâmetros incorretos$crlf" and
	 $dbHandle->disconnect() and die)
		if (!/\s*(\S+)\s*'(.*)'(.*)/);

	my $equipe_id     = $1;
	my $equipe_titulo = $2;
	my @informacao    = split(' ', $3);

	my $informacaoSize = scalar(@informacao);
	(print STDERR "$aplicativo\::Número de parâmetros incorretos$crlf" and
	 $dbHandle->disconnect() and die)
		if ($informacaoSize < 4);

	$equipe_titulo =~ s/^\s*(.*?)\s*$/$1/;
	(print STDERR "$aplicativo\::Título inválido$crlf" and
	 $dbHandle->disconnect() and die)
		if length($equipe_titulo)==0;

	(print STDERR "$aplicativo\::Número de parâmetros incorretos$crlf" and
	 $dbHandle->disconnect() and die)
		if ($informacaoSize != $informacao[2] + 3 +
		                       $informacao[$informacao[2] + 3]*2 + 1);

	$dbStatement1->bind_param(1, $equipe_id);
	$dbStatement1->bind_param(2, $equipe_titulo);
	$dbStatement1->bind_param(3, $informacao[0]);
	$dbStatement1->bind_param(4, $equipe_datainicio);

	$dbStatement1->execute()
		or (print STDERR "$aplicativo\::Falha de inserção: " .
		                 "$DBI::errstr$crlf" and
		    $dbHandle->disconnect() and die);

	$dbStatement2->bind_param(1, $equipe_id);
	$dbStatement2->bind_param(2, $equipe_disciplina);
	$dbStatement2->bind_param(3, $equipe_turma);
	$dbStatement2->bind_param(4, $equipe_datainicio);

	$dbStatement2->execute()
		or (print STDERR "$aplicativo\::Falha de inserção: " .
		                 "$DBI::errstr$crlf" and
		    $dbHandle->disconnect() and die);

	for (my $i=0; $i<$informacao[2]; $i++)
	{
		$dbStatement3->bind_param(1, $equipe_id);
		$dbStatement3->bind_param(2, $informacao[$i + 3]);
		$dbStatement3->bind_param(3, $equipe_datainicio);

		$dbStatement3->execute()
			or (print STDERR "$aplicativo\::Falha de inserção: " .
			                 "$DBI::errstr$crlf" and
			    $dbHandle->disconnect() and die);
	}

	my $deslocamento = $informacao[2] + 3;
	for (my $i=0; $i<$informacao[$deslocamento]; $i++)
	{
		(print STDERR "$aplicativo\::Orientação inválida$crlf" and
		 $dbHandle->disconnect() and die)
			if ($informacao[$deslocamento + $i*2 + 2] !~ /[01]/);

		$dbStatement4->bind_param(1, $equipe_id);
		$dbStatement4->bind_param(2, $informacao[$deslocamento + $i*2 + 1]);
		$dbStatement4->bind_param(3, $informacao[$deslocamento + $i*2 + 2]);
		$dbStatement4->bind_param(4, $equipe_datainicio);

		$dbStatement4->execute()
			or (print STDERR "$aplicativo\::Falha de inserção: " .
			                 "$DBI::errstr$crlf" and
			    $dbHandle->disconnect() and die);
	}

	$dbStatement5->bind_param(1, $equipe_id);
	$dbStatement5->bind_param(2, $informacao[1]);

	$dbStatement5->execute()
		or (print STDERR "$aplicativo\::Falha de inserção: " .
		                 "$DBI::errstr$crlf" and
		    $dbHandle->disconnect() and die);

	$dbHandle->commit() if $final_commit==0;

	print STDERR "Cód ($equipe_disciplina/$equipe_turma) <spc> 'tít' <spc> " . 
	             "id lider <spc> cód agregador <spc> núm alunos <spc> " .
	             "aluno ... <spc> núm orientadores <spc> orientador <spc> " .
	             "(0-Co/1-Ori) ... (CTRL+D aborta):";
}


# desconexão com banco de dados
$dbHandle->commit() if $final_commit==1;
$dbHandle->disconnect();
exit 0;
```
