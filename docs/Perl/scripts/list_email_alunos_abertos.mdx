```perl
#!/usr/bin/perl
use strict;
use warnings;
use DBI;
use Getopt::Std;

# -----------------------------------------------------------------------------
# Apresenta listagem de alunos matriculados.
# -----------------------------------------------------------------------------
$|++;
my $crlf = "\n";
my $aplicativo = $0;
my $data_formato_sql = "'%d/%m/%Y'";


# definição para acesso do banco (valores default)
my $host     = "localhost";
my $database = "DAELT";
my $usuario  = "";
my $senha    = "";

my $agregdiscip = "";


# Recuperação dos parâmetros de entrada (via shell)
my %opcoes=();
getopts("whu:p:a:", \%opcoes);

$crlf    = "\r\n"     if defined $opcoes{w};
$usuario = $opcoes{u} if defined $opcoes{u};
$senha   = $opcoes{p} if defined $opcoes{p};

$agregdiscip = $opcoes{a} if defined $opcoes{a};

if (defined $opcoes{h})
{
	print STDERR "$aplicativo [-w] [-u <usuario>] [-p <senha>] " .
	             "[-a agregdisc]$crlf";
	exit 0;
}


# apresentação do programa
print STDERR "$aplicativo\::Apresenta listagem de emails de alunos.$crlf";


# conexão com banco de dados
my ($dbHandle);
my ($dbStatement);
my (@dbLinha);

$dbHandle = DBI->connect("DBI:mysql:database=$database;host=$host",
                         $usuario, $senha,
                         {PrintError => 0, AutoCommit => 0})
	or die "$aplicativo\::Falha de conexão ao banco de dados: $DBI::errstr";

$dbStatement = $dbHandle->prepare(
	"SELECT A.EMAIL " .
	"FROM DAELT.ALUNOS A, DAELT.ALUNOS_DISCIPLINAS B, " .
	     "DAELT.DOMINIO_DISCIPLINAS C " .
	"WHERE A.ID = B.ID_ALUNO AND A.DATA_FIM IS NULL AND " .
	      "B.ID_DISCIPLINA = C.ID AND B.DATA_FIM IS NULL AND " .
	      "C.ID_AGREGADOR = ?");

$dbStatement->bind_param(1, $agregdiscip);
$dbStatement->execute();

while ( @dbLinha = $dbStatement->fetchrow_array )
{
	print "$dbLinha[0], "
}
print "$crlf";


# desconexão com banco de dados
$dbHandle->disconnect();
exit 0;
```
