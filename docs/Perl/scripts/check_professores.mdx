## Source Code

```perl
#!/usr/bin/perl
use strict;
use warnings;
use DBI;
use Getopt::Std;

# -----------------------------------------------------------------------------
# Verifica se existem professores
# -----------------------------------------------------------------------------
$|++;
my $crlf = "\n";
my $aplicativo = $0;


# definição para acesso do banco (valores default)
my $host     = "localhost";
my $database = "DAELT";
my $usuario  = "";
my $senha    = "";


# Recuperação dos parâmetros de entrada (via shell)
my %opcoes=();
getopts("whu:p:", \%opcoes);

$crlf    = "\r\n"     if defined $opcoes{w};
$usuario = $opcoes{u} if defined $opcoes{u};
$senha   = $opcoes{p} if defined $opcoes{p};

if (defined $opcoes{h})
{
	print STDERR "$aplicativo [-w] [-u <usuario>] [-p <senha>]$crlf";
	exit 0;
}


# apresentação do programa
print STDERR "$aplicativo\::Checa existência de nomes similares.$crlf";


# obtém nome do professor
print "Nome do professor:";
my $professor_nome = <STDIN>;
$professor_nome =~ s/^\s*(.*?)\s*$/$1/;
die "$aplicativo\::Digite um nome válido de professor"
	if length($professor_nome)==0;


# conexão com banco de dados
my ($dbHandle, $dbStatement1, $dbStatement2, $dbStatement3, $dbStatement4);
my (@dbLinha1, @dbLinha2);

$dbHandle = DBI->connect("DBI:mysql:database=$database;host=$host",
                         $usuario, $senha,
                         {PrintError => 0, AutoCommit => 0})
	or die "$aplicativo\::Falha de conexão ao banco de dados: $DBI::errstr";

$dbStatement1 = $dbHandle->prepare(
	"SELECT ID, NOME, EMAIL, EMAIL2, EFETIVO FROM DAELT.PROFESSORES " .
	"WHERE UPPER(NOME) LIKE UPPER(?) " .
	"ORDER BY NOME;");
$dbStatement2 = $dbHandle->prepare(
	"SELECT C.NOME " .
	"FROM DAELT.PROFESSORES A, DAELT.PROFESSORES_GRUPOS B, DAELT.GRUPOS C " .
	"WHERE A.ID = ? AND A.ID = B.ID_PROFESSOR AND B.ID_GRUPO = C.ID " .
	"ORDER BY C.NOME;");
$dbStatement3 = $dbHandle->prepare(
	"SELECT B.INTERESSE " .
	"FROM DAELT.PROFESSORES A, DAELT.PROFESSORES_INTERESSES B " .
	"WHERE A.ID = ? AND A.ID = B.ID_PROFESSOR " .
	"ORDER BY B.INTERESSE;");
$dbStatement4 = $dbHandle->prepare(
	"SELECT B.TEMA " .
	"FROM DAELT.PROFESSORES A, DAELT.PROFESSORES_TEMAS B " .
	"WHERE A.ID = ? AND A.ID = B.ID_PROFESSOR " .
	"ORDER BY B.TEMA;");

$dbStatement1->bind_param(1, "%$professor_nome%");
$dbStatement1->execute();

while ( @dbLinha1 = $dbStatement1->fetchrow_array )
{
	my $professor_id_saved = $dbLinha1[0];

	print "$dbLinha1[0]";
	print "\t$dbLinha1[1]" if ( defined $dbLinha1[1] );
	print "\t$dbLinha1[2]" if ( defined $dbLinha1[2] );
	print "\t$dbLinha1[3]" if ( defined $dbLinha1[3] );

	print "\tEFETIVO$crlf"   if ( $dbLinha1[4]!=0 );
	print "\tVISITANTE$crlf" if ( $dbLinha1[4]==0 );

	$dbStatement2->bind_param(1, $professor_id_saved);
	$dbStatement2->execute();

	while ( @dbLinha2 = $dbStatement2->fetchrow_array )
	{
		print "\t1\t$dbLinha2[0]$crlf";
	}

	$dbStatement3->bind_param(1, $professor_id_saved);
	$dbStatement3->execute();

	while ( @dbLinha2 = $dbStatement3->fetchrow_array )
	{
		print "\t2\t$dbLinha2[0]$crlf";
	}

	$dbStatement4->bind_param(1, $professor_id_saved);
	$dbStatement4->execute();

	while ( @dbLinha2 = $dbStatement4->fetchrow_array )
	{
		print "\t3\t$dbLinha2[0]$crlf";
	}
	print "$crlf";
}


# desconexão com banco de dados
$dbHandle->disconnect();
exit 0;
```
