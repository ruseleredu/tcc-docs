#!/usr/bin/perl
use strict;
use warnings;
use DBI;
use Getopt::Std;
use POSIX qw(strftime);

# -----------------------------------------------------------------------------
# Insere disciplinas.
# -----------------------------------------------------------------------------
$|++;
my $crlf = "\n";
my $aplicativo = $0;


# definição para acesso do banco (valores default)
my $host         = "localhost";
my $database     = "DAELT";
my $usuario      = "";
my $senha        = "";
my $final_commit = 0;


# Recuperação dos parâmetros de entrada (via shell)
my %opcoes=();
getopts("whu:p:", \%opcoes);

$crlf         = "\r\n"     if defined $opcoes{w};
$usuario      = $opcoes{u} if defined $opcoes{u};
$senha        = $opcoes{p} if defined $opcoes{p};
$final_commit = 1          if defined $opcoes{f};

if (defined $opcoes{h})
{
	print STDERR "$aplicativo [-w] [-u <usuario>] [-p <senha>] [-f]$crlf";
	exit 0;
}


# apresentação do programa
print STDERR "$aplicativo\::Insere disciplinas.\n";


# conexão com banco de dados
my ($dbHandle, $dbStatement);

$dbHandle = DBI->connect("DBI:mysql:database=$database;host=$host",
                         $usuario, $senha,
                         {PrintError => 0, AutoCommit => 0})
	or die "$aplicativo\::Falha de conexão ao banco de dados: $DBI::errstr";

$dbStatement = $dbHandle->prepare(
	"INSERT INTO DAELT.DOMINIO_DISCIPLINAS " .
	"(ID, ID_RESPONSAVEL, ID_COORDENADOR, ID_AGREGADOR, NOME) " .
	"VALUES (?, ?, ?, ?, ?);");


# definição de data de início para TODAS as disciplinas
my $disciplina_datainicio = strftime "%Y-%02m-%02d", localtime;


# obtém nome da disciplina
print STDERR "Cód <spc> 'Nome' <spc> id responsavel <spc> " .
             "id coordenador <spc> agregador:";
while (<STDIN>)
{
	(print STDERR "$aplicativo\::Entrada incorreta$crlf" and
	 $dbHandle->disconnect() and die)
		if (!/\s*(\S+)\s*'(.*)'(.*)/);

	my $disciplina_id   = $1;
	my $disciplina_nome = $2;
	my @informacao      = split(' ', $3);

	my $informacaoSize = scalar(@informacao);
	(print STDERR "$aplicativo\::Número de parâmetros incorretos$crlf" and
	 $dbHandle->disconnect() and die)
		if ($informacaoSize != 3);

	$disciplina_nome =~ s/^\s*(.*?)\s*$/$1/;
	(print STDERR "$aplicativo\::Nome de disciplina inválido$crlf" and
	 $dbHandle->disconnect() and die)
		if length($disciplina_nome)==0;

	$dbStatement->bind_param(1, $disciplina_id);
	$dbStatement->bind_param(2, $informacao[0]);
	$dbStatement->bind_param(3, $informacao[1]);
	$dbStatement->bind_param(4, $informacao[2]);
	$dbStatement->bind_param(5, $disciplina_nome);

	$dbStatement->execute()
		or (print STDERR "$aplicativo\::Falha de inserção: " .
		                 "$DBI::errstr$crlf" and
		    $dbHandle->disconnect() and die);

	$dbHandle->commit() if $final_commit==0;

	print STDERR "Cód <spc> 'Nome' <spc> id responsavel <spc> " .
	             "id coordenador <spc> agregador:";
}


# desconexão com banco de dados
$dbHandle->commit() if $final_commit==1;
$dbHandle->disconnect();
exit 0;
